## 📖 详细描述

### 问题说明

当前记忆系统像一个**图书馆**（按时间归档），但不是一个**大脑**（会遗忘）。

**核心缺失**：
1. 不记录记忆的最后访问时间
2. 不追踪记忆的访问频率
3. 没有记忆重要性/强度评分
4. 没有随时间衰减的逻辑
5. 没有基于重要性的分级存储

### 人类记忆类比

| 记忆类型 | 特征 | 对应实现 |
|----------|------|----------|
| **工作记忆** | 当前正在思考，极活跃 | 待实现 - Active 状态 |
| **短期记忆** | 最近访问，易回忆 | 待实现 - Cold 状态 |
| **长期记忆** | 重要/经常访问，持久 | 待实现 - 高 Importance 值 |
| **遗忘边缘** | 很少访问，即将消失 | 待实现 - Fading 状态 |
| **归档记忆** | 压缩摘要，永久保存 | 已实现 - IsSummary=true |

#### 3.1 创建 MemoryFadeService

**文件**: `src/SiliconLife.Core/SiliconBeing/MemoryFadeService.cs` (新建)

```csharp
// Copyright (c) 2026 Hoshino Kennji

using System;

namespace SiliconLife.Collective;

/// <summary>
/// Manages the forgetting/decay mechanism for memory entries.
/// Runs periodically to update memory importance based on time since last access.
/// </summary>
public class MemoryFadeService : TickObject
{
    private static readonly ILogger _logger = LogManager.Instance.GetLogger<MemoryFadeService>();
    private readonly Memory _memory;
    private readonly double _fadeThreshold;      // Importance < 0.1 → Fading
    private readonly double _coldThreshold;     // Importance < 0.3 → Cold
    private readonly double _activeThreshold;   // Importance >= 0.3 → Active
    private readonly TimeSpan _tickInterval;

    /// <summary>
    /// Creates a new MemoryFadeService.
    /// </summary>
    /// <param name="memory">The memory system to manage</param>
    /// <param name="tickInterval">How often to run decay calculations (default: 1 hour)</param>
    /// <param name="fadeThreshold">Importance threshold for Fading state (default: 0.1)</param>
    /// <param name="coldThreshold">Importance threshold for Cold state (default: 0.3)</param>
    public MemoryFadeService(
        Memory memory,
        TimeSpan? tickInterval = null,
        double fadeThreshold = 0.1,
        double coldThreshold = 0.3)
        : base(tickInterval ?? TimeSpan.FromHours(1), autoRegister: true)
    {
        _memory = memory;
        _fadeThreshold = fadeThreshold;
        _coldThreshold = coldThreshold;
        _activeThreshold = coldThreshold;
        _tickInterval = tickInterval ?? TimeSpan.FromHours(1);
        
        _logger.Info(null, "MemoryFadeService initialized, interval={0}", _tickInterval);
    }

    /// <summary>
    /// Executes the decay calculation for all memory entries.
    /// </summary>
    protected override void OnTick(TimeSpan deltaTime)
    {
        _logger.Debug(null, "Running memory decay calculation...");
        
        var allEntries = _memory.QueryAll();
        int updatedCount = 0;
        int stateChangeCount = 0;
        
        foreach (var entry in allEntries)
        {
            // Skip archived summaries (they are permanently preserved)
            if (entry.IsSummary)
            {
                continue;
            }

            // Calculate days since last access
            double daysSinceAccess = (DateTime.Now - entry.LastAccessTime).TotalDays;
            if (daysSinceAccess < 0) daysSinceAccess = 0;
            
            // Apply exponential decay
            double oldImportance = entry.Importance;
            entry.Importance = Math.Max(0.0, entry.Importance * Math.Exp(-entry.DecayRate * daysSinceAccess));
            
            // Update state based on new importance
            MemoryState oldState = entry.State;
            entry.State = CalculateState(entry.Importance);
            
            if (oldState != entry.State)
            {
                stateChangeCount++;
                _logger.Debug(null, "Memory {0} state changed: {1} → {2}, importance={3:F3}", 
                    entry.Id, oldState, entry.State, entry.Importance);
            }
            
            // Only save if importance changed significantly (>1%)
            if (Math.Abs(oldImportance - entry.Importance) > 0.01 || oldState != entry.State)
            {
                _memory.Save(entry);
                updatedCount++;
            }
        }
        
        _logger.Info(null, "Memory decay complete: {0} entries processed, {1} updated, {2} state changes", 
            allEntries.Count, updatedCount, stateChangeCount);
    }

    /// <summary>
    /// Calculates the memory state based on importance value.
    /// </summary>
    private MemoryState CalculateState(double importance)
    {
        if (importance >= _activeThreshold)
            return MemoryState.Active;
        else if (importance >= _fadeThreshold)
            return MemoryState.Cold;
        else
            return MemoryState.Fading;
    }

    /// <summary>
    /// Manually boosts a memory's importance (e.g., when user marks as important).
    /// </summary>
    public void BoostImportance(Guid memoryId, double boostAmount = 0.2)
    {
        var entries = _memory.QueryAll();
        var entry = entries.FirstOrDefault(e => e.Id == memoryId);
        if (entry != null)
        {
            entry.Importance = Math.Min(1.0, entry.Importance + boostAmount);
            entry.State = CalculateState(entry.Importance);
            _memory.Save(entry);
            _logger.Info(null, "Boosted memory {0} importance to {1:F3}", memoryId, entry.Importance);
        }
    }
}
```

/// <summary>
/// Gets memory entries filtered by state.
/// </summary>
public List<MemoryEntry> QueryByState(MemoryState state, int count = 0)
{
    var all = QueryAll();
    var filtered = all.Where(e => e.State == state).OrderByDescending(e => e.Importance);
    return (count > 0 ? filtered.Take(count) : filtered).ToList();
}

/// <summary>
/// Gets statistics about memory states.
/// </summary>
public MemoryStateStatistics GetStateStatistics()
{
    var all = QueryAll();
    return new MemoryStateStatistics
    {
        TotalEntries = all.Count,
        ActiveCount = all.Count(e => e.State == MemoryState.Active),
        ColdCount = all.Count(e => e.State == MemoryState.Cold),
        FadingCount = all.Count(e => e.State == MemoryState.Fading),
        ArchivedCount = all.Count(e => e.IsSummary),
        AverageImportance = all.Average(e => e.Importance),
        OldestActiveEntry = all.Where(e => e.State == MemoryState.Active).MinBy(e => e.Timestamp)?.Timestamp,
    };
}

/// <summary>
/// Archives fading memories (creates compressed summary and marks originals for deletion).
/// </summary>
public int ArchiveFadingMemories()
{
    var fading = QueryByState(MemoryState.Fading);
    int archivedCount = 0;
    
    foreach (var group in fading.GroupBy(e => e.Timestamp.Date))
    {
        // Create summary for this day's fading memories
        var summary = string.Join("\n", group.Select(e => e.Content).Take(10));
        AddSummary(new IncompleteDate(group.Key.Year, group.Key.Month, group.Key.Day), summary);
        archivedCount++;
    }
    
    _logger.Info(null, "Archived {0} fading memories into {1} summaries", fading.Count, archivedCount);
    return archivedCount;
}
